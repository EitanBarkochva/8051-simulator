💻 שפת C למיקרו-בקר 8051

מדריך שפת C לבקר C8051F380 של Silicon Labs (משפחת 8051), בסביבת הפיתוח Keil µVision. כאן מרוכזים טיפוסי הנתונים, האופרטורים, מבני התנאי והלולאה, המערכים, הפונקציות ושגרות הפסיקה — עם דוגמאות קוד.

תוכן העניינים

מבנה תוכנית טיפוסי נתונים אזורי זיכרון קדם-מעבד אופרטורים מבני תנאי לולאות מערכים פונקציות פסיקות (ISR) השהיות

מבנה תוכנית בסיסי

כל תוכנית מתחילה בהכללת קובצי ההגדרות של הבקר, ולאחר מכן פונקציית main שבתוכה לולאה אינסופית while(1) שבה רץ הקוד שלכם שוב ושוב.

#include "compiler_defs.h"
#include "C8051F380_defs.h"

void Init_Device(void);                 // אתחול הבקר (כיבוי ה-Watchdog, הגדרת שעון ופינים)
void delay_ms(unsigned int ms);
void delay_us(unsigned int us);

void main()
{
    Init_Device();                      // חובה בתחילת התוכנית
    while (1)
    {
        // הקוד שלך כאן — רץ בלולאה אינסופית
    }
}
שים לב: את הביטוי Init_Device() חובה לקרוא בתחילת main — הוא מכבה את ה-Watchdog ומגדיר את השעון והפינים של הבקר.

אזורי זיכרון (Memory Areas)

בבקר 8051 קובעים היכן יישמר משתנה באמצעות מילת מפתח לפני שם הטיפוס:

מילת מפתחתיאורדוגמה
dataRAM פנימי עם כתובת ישירה (מהיר)unsigned int data dnum;
idataRAM פנימי עם כתובת עקיפהint idata inum;
xdataRAM חיצוניunsigned char xdata xnum;
codeזיכרון התוכנית (קבועים, לקריאה בלבד)unsigned char code cnum = 0xAA;

ניתן למקם משתנה בכתובת מוחלטת בעזרת _at_:

unsigned char xdata xnum _at_ 0x8000;   // המשתנה ימוקם בכתובת 0x8000

אופרטורים ביטיים (Bitwise)

אופרטורפעולה
&AND ברמת הביט
|OR ברמת הביט
^XOR — או בלעדי
~NOT — היפוך כל הביטים
<<הזזה שמאלה (Shift Left)

טיפ: אופרטורים ביטיים שימושיים מאוד לשליטה בפינים בודדים — P1 |= 0x01; מדליק את ביט 0, P1 &= ~0x01; מכבה אותו, ו-P1 ^= 0x01; הופך את מצבו.

לולאות (Iteration)

סוגתחביר
whilewhile (expression) statements
do-whiledo statements while (condition);
forfor (init; condition; increase) statements
// while — בודק תנאי לפני כל מחזור
while (n > 0) {
    P1 = n;
    n--;
}

// do-while — מבצע לפחות פעם אחת, בודק תנאי בסוף
do {
    n = P1;
} while (n != 0);

// for — אתחול, תנאי, וקידום במשתנה מונה
for (i = 0; i < 10; i++) {
    P1 = i;
}

מערכים (Arrays)

פעולהדוגמה
הצהרהchar arr[5];
הצהרה עם אתחולint arr[5] = {3, 5, 7, -1, 14};
מערך דו-ממדיint arr[3][5];
גישה לאיברarr[1][3]

איברי המערך ממוספרים החל מ-0. כך למשל arr[0] הוא האיבר הראשון ו-arr[4] הוא החמישי במערך בגודל 5.

פונקציות (Functions)

פונקציה ללא טיפוס וללא ארגומנט

void OutData(void)
{
    P1 = 0xAB;
}

void main()
{
    OutData();
}

פונקציה עם ארגומנטים (ללא ערך מוחזר)

void OutData(unsigned char a, unsigned char b)
{
    P0 = a;
    P1 = b;
}

void main()
{
    OutData(0x0F, 0xF0);
}

פונקציה עם טיפוס מוחזר וארגומנט

unsigned char InOutData(unsigned char a)
{
    P1 = a;
    return P0;          // מחזירה ערך
}

void main()
{
    unsigned char r;
    r = InOutData(63);
    P2 = r;
}

parameter — פרמטר (ארגומנט); type — טיפוס הערך המוחזר; return — מחזיר ערך מהפונקציה.

שגרות פסיקה (Interrupt Service Routines)

פסיקה (Interrupt) היא אירוע שמפסיק את הריצה הרגילה ומקפיץ את הבקר לשגרה ייעודית. מגדירים שגרת פסיקה עם המילה interrupt ומספר הפסיקה:

void int2sub() interrupt 2 {
    // קוד הפסיקה — ירוץ כשפסיקה מספר 2 מתרחשת
}
מקור הפסיקהמספרכתובת וקטור

External Interrupt 0 (INT0)00x0003
Timer 0 Overflow10x000B
External Interrupt 1 (INT1)20x0013
Timer 1 Overflow30x001B
UART040x0023
Timer 2 Overflow50x002B
SPI060x0033
SMBus (SMB0)70x003B
ADC0 Conversion Complete100x0053
להפעלת פסיקה צריך גם לאפשר אותה ברגיסטר IE (Interrupt Enable) ולהדליק את ביט ה-EA (אישור פסיקות גלובלי). פירוט מלא בעמוד הרגיסטרים.

השהיות (Delay)

לבקר יש פונקציות השהיה מובנות ליצירת עיכוב מדויק (למשל להבהוב נורה):

void delay_ms(unsigned int ms);   // השהיה באלפיות שנייה
void delay_us(unsigned int us);   // השהיה במיליוניות שנייה

// דוגמה — הבהוב נורה על P1.0
while (1) {
    P1 = 0x01;
    delay_ms(500);
    P1 = 0x00;
    delay_ms(500);
}
